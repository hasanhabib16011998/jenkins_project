pipeline {
    agent any

    parameters {
        string(name: 'IMAGE_TAG', defaultValue: '', description: 'Docker image tag passed from the triggering pipeline')
    }

    environment {
        DOCKERHUB_USER = 'hasanhabib16011998'
        FRONTEND_APP = "jenkins-project"
        FRONTEND_IMAGE = "${DOCKERHUB_USER}/${FRONTEND_APP}"
    }

    stages {
        stage('Use Docker Image') {
            steps {
                echo "Using Docker image: ${FRONTEND_IMAGE}:${IMAGE_TAG}"
            }
        }
    }

    post {
        success {
            emailext body: "Hi, the pipeline 2 has been built successfully. ${FRONTEND_IMAGE}:${IMAGE_TAG} has been pushed to DockerHub.",
                     subject: 'Pipeline_2 Build Success',
                     to: 'jinaj50765@opposir.com'
        }
        failure {
            emailext body: "Hi, the pipeline 2 build has failed. Please check the Jenkins logs for more details.",
                     subject: 'Pipeline_2 Build Failure',
                     to: 'jinaj50765@opposir.com'
        }
    }
}
